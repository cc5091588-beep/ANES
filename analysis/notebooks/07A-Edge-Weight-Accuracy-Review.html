1. Purpose and analysis boundary

This notebook performs a read-only accuracy review of the five formal edge-weight bootstrap results produced in Step 07.

It does not estimate networks, rerun bootstrap resampling, calculate centrality, conduct case-dropping bootstrap, perform NCT, analyse education groups, or interpret political change.

Bootstrap intervals are treated as indicators of estimation accuracy. Whether an interval includes zero is not treated as a formal significance test.

wave_summary <- do.call(
  rbind,
  lapply(
    split(
      edge_accuracy,
      edge_accuracy$Wave
    ),
    function(wave_data) {
      data.frame(
        Wave = unique(wave_data$Wave),
        Edges = nrow(wave_data),
        Retained_edges = sum(
          wave_data$Retained_under_EBICglasso
        ),
        All_edges_percentile_interval_excluding_zero =
          sum(
            !wave_data$
              Percentile_interval_includes_zero
          ),
        Retained_edges_percentile_interval_excluding_zero =
          sum(
            wave_data$Retained_under_EBICglasso &
            !wave_data$
              Percentile_interval_includes_zero
          ),
        Median_percentile_interval_width =
          median(
            wave_data$
              Percentile_interval_width
          ),
        Maximum_percentile_interval_width =
          max(
            wave_data$
              Percentile_interval_width
          ),
        Median_bootstrap_zero_proportion =
          median(
            wave_data$
              Bootstrap_zero_proportion
          ),
        Maximum_absolute_mean_point_difference =
          max(
            abs(
              wave_data$
                Bootstrap_mean_minus_point
            )
          ),
        Retained_edges_with_reproduced_sign =
          sum(
            wave_data$
              Retained_edge_sign_reproduced,
            na.rm = TRUE
          ),
        stringsAsFactors = FALSE
      )
    }
  )
)

figure_files <- character(
  length(expected_waves)
)

for (wave_index in seq_along(expected_waves)) {
  current_wave <- expected_waves[wave_index]

  wave_data <- edge_accuracy[
    edge_accuracy$Wave == current_wave,
  ]

  wave_data <- wave_data[
    order(wave_data$Point_edge),
  ]

  y_position <- seq_len(
    nrow(wave_data)
  )

  point_colours <- ifelse(
    wave_data$Retained_under_EBICglasso,
    "#0072B2",
    "#999999"
  )

  x_limits <- range(
    c(
      wave_data$Percentile_2_5,
      wave_data$Percentile_97_5
    ),
    finite = TRUE
  )

  x_padding <- diff(x_limits) * 0.05

  if (!is.finite(x_padding) || x_padding == 0) {
    x_padding <- 0.05
  }

  figure_file <- file.path(
    formal_figure_dir,
    paste0(
      "07A_edge_accuracy_",
      current_wave,
      ".png"
    )
  )

  png(
    filename = figure_file,
    width = 2000,
    height = 1700,
    res = 180
  )

  par(
    mar = c(
      5,
      12,
      4,
      2
    ) + 0.1
  )

  plot(
    wave_data$Point_edge,
    y_position,
    xlim = c(
      x_limits[1] - x_padding,
      x_limits[2] + x_padding
    ),
    ylim = c(
      0.5,
      nrow(wave_data) + 0.5
    ),
    yaxt = "n",
    pch = 19,
    col = point_colours,
    xlab = "Regularised partial-correlation edge weight",
    ylab = "",
    main = paste(
      current_wave,
      "edge-weight accuracy"
    )
  )

  segments(
    x0 = wave_data$Percentile_2_5,
    y0 = y_position,
    x1 = wave_data$Percentile_97_5,
    y1 = y_position,
    col = point_colours,
    lwd = 1.5
  )

  points(
    wave_data$Point_edge,
    y_position,
    pch = 19,
    col = point_colours
  )

  axis(
    side = 2,
    at = y_position,
    labels = wave_data$Edge_ID,
    las = 2,
    cex.axis = 0.65
  )

  abline(
    v = 0,
    lty = 2,
    col = "#444444"
  )

  legend(
    "bottomright",
    legend = c(
      "Retained under EBICglasso",
      "Not retained under EBICglasso"
    ),
    col = c(
      "#0072B2",
      "#999999"
    ),
    pch = 19,
    bty = "n",
    cex = 0.8
  )

  dev.off()

  figure_files[wave_index] <- figure_file
